cmake_minimum_required(VERSION 3.16)
if(NOT DEFINED BOARD)
  set(BOARD metalio_claw4 CACHE STRING "metalio_claw4 or m5stack_tab5")
endif()
if(NOT BOARD MATCHES "^(metalio_claw4|m5stack_tab5)$")
  message(FATAL_ERROR "Unsupported BOARD=${BOARD}")
endif()
set(IDF_TARGET esp32p4 CACHE STRING "Target")
set(SDKCONFIG "${CMAKE_BINARY_DIR}/sdkconfig" CACHE FILEPATH "Per-board configuration")
set(SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.defaults;${CMAKE_CURRENT_LIST_DIR}/boards/${BOARD}.defaults")
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(p4_native_display VERSION 0.1.0)
